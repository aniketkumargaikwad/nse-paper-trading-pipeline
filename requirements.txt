# ---------------------------------------------------------------------------
# Pinned dependencies for the paper-trading research pipeline.
#
# Versions are pinned exactly so that GitHub Actions, Streamlit Cloud, and
# your local machine all run identical code. If you upgrade anything, upgrade
# deliberately (one package at a time) and re-run the test suite.
#
# NOTE: These pins were chosen as the latest stable releases known at build
# time (early 2026). Verify against PyPI if an install fails.
# ---------------------------------------------------------------------------

# --- Market data providers --------------------------------------------------
# yfinance: the DEFAULT, FREE data source (no account, no API key, no daily
# login). Not pinned to an exact version on purpose: it is an unofficial
# Yahoo scraper that must be upgradable quickly when Yahoo changes its API.
yfinance>=0.2.65

# Kite Connect (Zerodha): OPTIONAL, only used when DATA_PROVIDER=kite.
# Requires the paid "Connect" plan — the free Personal tier excludes the
# historical-data API this project runs on.
kiteconnect==5.0.1

# TOTP generation for unattended Dhan token renewal (no daily login).
pyotp==2.9.0

# Direct REST calls to Dhan, carrying API credentials. requests also arrives
# transitively via kiteconnect, streamlit, and yfinance, but it is pinned
# explicitly here rather than left to whatever those resolve to. The pin must
# be at least 2.32.4: earlier 2.32.x releases carry CVE-2024-47081, a
# URL-parsing flaw that can leak .netrc credentials to a malicious host via
# crafted redirects — a real risk for a client that carries API credentials.
requests==2.32.4

# Supabase Python client — the only persistent state store.
supabase==2.15.0

# Data wrangling for candles and indicators.
pandas==2.2.3
# numpy 2.x, NOT <2.0. The old pin was 1.26.4 "for maximum wheel
# compatibility on free tiers" - a line that stopped being true the moment
# pandas-ta-classic was added below, because it requires numpy>=2.0.0. Those
# two cannot both be satisfied on ANY Python version, so this file could not
# be installed from scratch anywhere; it survived only on machines where a
# working numpy already sat in the venv. Railway hit it on a clean build and
# stopped at ResolutionImpossible.
#
# 2.4.6 is also simply what runs: the suite has been passing against it all
# along, while this file named a version nobody had installed.
#
# Adding a dependency can therefore invalidate a pin far from it. Re-resolve
# from empty after touching this file:
#     pip install --dry-run --ignore-installed -r requirements.txt
numpy==2.4.6

# Parquet engine. pandas.read_parquet/to_parquet delegate to this, so
# CANDLE_STORE=parquet cannot read or write a single candle without it. It
# was missing from this file entirely while being installed locally, which
# is the kind of gap that only shows up on a fresh machine.
pyarrow==25.0.0

# Strategy config parsing.
PyYAML==6.0.2

# Local .env loading (GitHub Actions injects real env vars instead).
python-dotenv==1.0.1

# Timezone database — REQUIRED on Windows for zoneinfo("Asia/Kolkata") to work.
# (Linux/macOS have a system tz database; Windows does not.)
tzdata==2025.2

# Dashboard (Streamlit Community Cloud reads this same root requirements.txt).
# Trade-off: including it here slows CI installs by ~30s but keeps ONE file
# for a beginner to manage instead of two.
streamlit==1.41.1

# Testing.
pytest==8.3.4

# Independent indicator libraries, used ONLY to check our own maths against
# somebody else's — see tests/test_indicators_vs_reference.py. An RSI that
# looks right but disagrees with every chart the owner reads is not an RSI,
# and unrelated authors making the same mistake is unlikely.
#
# The test skips itself if these are absent, so a deploy that installs only
# the runtime packages still works. They are pinned here so the check runs
# the same way on any machine.
pandas-ta-classic==0.6.52
ta==0.11.0
